   \[
   (r_1 \operatorname{cis} \theta_1)(r_2 \operatorname{cis} \theta_2) = (r_1 \cdot r_2) \operatorname{cis} (\theta_1 + \theta_2)
   \]

   \[
   -2 \operatorname{cis} 37^\circ = 2 \operatorname{cis} (37^\circ + 180^\circ) = 2 \operatorname{cis} 217^\circ

   \[
   (3 \operatorname{cis} 18^\circ)(2 \operatorname{cis} 217^\circ) = (3 \cdot 2) \operatorname{cis} (18^\circ + 217^\circ) = 6 \operatorname{cis} 235^\circ
