def incenter(x1, y1, x2, y2, x3, y3):
    a = sp.sqrt((x2 - x3)**2 + (y2 - y3)**2)
    b = sp.sqrt((x1 - x3)**2 + (y1 - y3)**2)
    c = sp.sqrt((x1 - x2)**2 + (y1 - y2)**2)
