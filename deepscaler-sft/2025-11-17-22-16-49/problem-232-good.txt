```python
import numpy as np

# We will represent a 3x3x3 Rubik's cube as a 3D numpy array.
cube = np.zeros((3, 3, 3), dtype=bool)

# Function to check if two cells share a common vertex.
def share_common_vertex(pos1, pos2):
    x1, y1, z1 = pos1
    x2, y2, z2 = pos2
    return (abs(x1 - x2) + abs(y1 - y2) + abs(z1 - z2)) == 1

# Now let's try to place the maximum number of cells such that no two marked cells share a common vertex.
marked_cells = []

# Try to place cells one by one and check the condition
for i in range(3):
    for j in range(3):
        for k in range(3):
            if all(not share_common_vertex((i, j, k), (x, y, z)) for x, y, z in marked_cells):
                marked_cells.append((i, j, k))
                cube[i, j, k] = True

# We need to get all surface cells of the 3x3x3 Rubik's Cube.
surface_cells = [(i, j, k) for i in range(3) for j in range(3) for k in range(3) if cube[i, j, k]]

print((len(marked_cells), surface_cells))
```
