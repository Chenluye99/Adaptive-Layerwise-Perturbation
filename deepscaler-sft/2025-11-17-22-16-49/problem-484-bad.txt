\[
\overrightarrow{a} \cdot \overrightarrow{a} + \overrightarrow{a} \cdot \overrightarrow{b} = 0

1 + \overrightarrow{a} \cdot \overrightarrow{b} = 0 \implies \overrightarrow{a} \cdot \overrightarrow{b} = -1
