def to_base_11(n):
    """Convert a number n to its base 11 representation as a list of digits."""
    if n == 0:
        return [0]
    
    base_11_digits = []
    while n > 0:
        base_11_digits.append(n % 11)
        n = n // 11
    
    # The digits are generated in reverse order, so we need to reverse the list
    base_11_digits.reverse()
    return base_11_digits

count = 0

        if len(base_11_rep) > 1 and base_11_rep[0] == 0:
            continue

        leftmost_digit = base_11_rep[0]
        rightmost_digit = base_11_rep[-1]
