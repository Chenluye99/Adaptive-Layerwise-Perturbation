import numpy as np
from scipy.optimize import minimize

# Define the function to minimize: x - y
def objective(vars):
    x, y = vars
    return x - y

# Define the constraint: 1/(x + 2) + 1/(1 - y) - 1/6 = 0
def constraint(vars):
    x, y = vars
    return 1 / (x + 2) + 1 / (1 - y) - 1 / 6

# Define the bounds for x and y based on the conditions x > 0 and y < 0
bounds = [(0, None), (-1, 0)]

# Initial guess for x and y
initial_guess = [1, -0.5]

# Define constraints in the form required by `minimize`
cons = ({'type': 'eq', 'fun': constraint})

# Perform the optimization
solution = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=cons)

x_opt, y_opt = solution.x
min_difference = solution.fun

print(((x_opt, y_opt, min_difference)))
