```python
import sympy

def get_distinct_prime_factors(n):
    """Get the distinct prime factors of n."""
    return list(sympy.primefactors(n))

def is_prime(k):
    """Check if a number k is prime."""
    return sympy.isprime(k)

def is_primer(n):
    """Check if a number n has a prime number of distinct prime factors."""
    distinct_prime_factors = get_distinct_prime_factors(n)
    num_distinct_prime_factors = len(distinct_prime_factors)
    return is_prime(num_distinct_prime_factors)

def is_primest(n):
    """Check if a number has a primest number of distinct primer factors."""
    distinct_primer_factors = []
    for factor in get_distinct_prime_factors(n):
        if is_primer(factor):
            distinct_primer_factors.append(factor)
    
    num_distinct_primer_factors = len(distinct_primer_factors)
    
    return is_primer(num_distinct_primer_factors)

def is_prime_minister(n):
    """Check if a number has a primest number of distinct primest factors."""
    distinct_primest_factors = []
    for factor in get_distinct_prime_factors(n):
        if is_primest(factor):
            distinct_primest_factors.append(factor)
    
    num_distinct_primest_factors = len(distinct_primest_factors)
    
    return is_primest(num_distinct_primest_factors)

# Find the smallest prime-minister number
N = 2
while True:
    if is_prime_minister(N):
```
