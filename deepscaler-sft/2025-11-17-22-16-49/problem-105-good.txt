# Calculate the midpoints R1, R3, R5, R7
R1 = ((vertices[0][0] + vertices[1][0]) / 2, (vertices[0][1] + vertices[1][1]) / 2)

R7 = ((vertices[6][0] + vertices[7][0]) / 2, (vertices[6][1] + vertices[7][1]) / 2)

# List of midpoints
R_points = [R1, R3, R5, R7]

# Function to compute the area using the shoelace formula
def shoelace_formula(coords):
    n = len(coords)
    sum1 = sum(coords[i][0] * coords[(i + 1) % n][1] for i in range(n))
    sum2 = sum(coords[i][1] * coords[(i + 1) % n][0] for i in range(n))
    area = 0.5 * abs(sum1 - sum2)
    return area
