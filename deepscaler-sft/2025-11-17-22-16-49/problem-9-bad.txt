import sympy as sp

# Define the function f(x)
x = sp.symbols('x')
a = sp.symbols('a', real=True)
f = a**2 * sp.sin(2 * x) + (a - 2) * sp.cos(2 * x)

# Substitute x = -pi/8 into f(x)

# To ensure symmetry about x = -pi/8, f(-pi/8 + h) should equal f(-pi/8 - h)

f_at_x0_plus_h = f.subs(x, x0 + h)
f_at_x0_minus_h = f.subs(x, x0 - h)

# Set f(-pi/8 + h) equal to f(-pi/8 - h) and solve for a

solutions = sp.solve(eq, a)

# Filter solutions to ensure a < 0
solutions = [sol for sol in solutions if sol < 0]

# We get the value of a that satisfies the condition, and then we substitute it back to find the maximum value of f(x)
a_value = solutions[0]

# Substitute a back into f(x) and find its maximum value
f_substituted = f.subs(a, a_value)

# To find the maximum value of f(x), we take the derivative and set it to 0 to find the critical points

critical_points = sp.solve(f_prime, x)

# Evaluate f(x) at critical points to get the maximum value
