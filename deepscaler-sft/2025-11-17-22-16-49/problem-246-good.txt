import numpy as np

def place_rooks(n):
    board = [[0 for _ in range(8)] for _ in range(8)]
    
    # Place white rooks
    for i in range(n):
        board[i][i] = 'W'
    
    # Place black rooks such that no black and white rook attack each other
    placed_black = set()
    for i in range(n):
        for j in range(8):
            if (i, j) not in placed_black:
                if all(board[i][k] != 'B' for k in range(8)) and all(board[k][j] != 'W' for k in range(8)):
                    board[i][j] = 'B'
                    placed_black.add((i, j))
                    break
    
    # Check if the placement is valid
    for i in range(8):
        for j in range(8):
            piece = board[i][j]
            if piece == 'W':
                if any(board[i][k] == 'B' for k in range(8)) or any(board[k][j] == 'B' for k in range(8)):
                    return False
            elif piece == 'B':
                if any(board[i][k] == 'W' for k in range(8)) or any(board[k][j] == 'W' for k in range(8)):
                    return False
    
    return True

max_n = 0
for n in range(1, 9):
    if place_rooks(n):
        max_n = n

print(max_n)
