```python
import sympy as sp

# Define variables
x = sp.symbols('x')
a, b, c = sp.symbols('a b c')

# Define P(x)
P = a * x**2 + b * x + c

# Compute P(P(x))
P_of_P = P.subs(x, P)

# Compute (x^2 + x + 1) * P(x)
right_hand_side = (x**2 + x + 1) * P

# Expand both expressions
lhs = sp.expand(P_of_P)
rhs = sp.expand(right_hand_side)

# Get the list of coefficients for both sides by comparing the powers of x
coeff_lhs = [lhs.coeff(x, i) for i in range(0, 6)]
coeff_rhs = [rhs.coeff(x, i) for i in range(0, 6)]

# Set up the system of equations by equating the corresponding coefficients
equations = [sp.Eq(coeff_lhs[i], coeff_rhs[i]) for i in range(6) if coeff_lhs[i] != 0 or coeff_rhs[i] != 0]

# Solve the system of equations for a, b, c
solution = sp.solve(equations, (a, b, c))

print(solution)
```
