def construct_matrix(n):
    A = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            A[i, j] = n - abs(i - j)
    return A

# Find the smallest eigenvalue of the matrix A for increasing n and check for convergence.
eigenvalues_list = []

for n in range(1, 11):
    A = construct_matrix(n)
    eigenvalues = np.linalg.eigvals(A)
