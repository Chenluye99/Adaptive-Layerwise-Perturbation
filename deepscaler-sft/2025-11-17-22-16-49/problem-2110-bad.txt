   - The slope between two points \((x_1, y_1)\) and \((x_2, y_2)\) is given by \(\text{slope} = \frac{y_2 - y_1}{x_2 - x_1}\).

import sympy as sp

# List of points
points = [(2, 2), (9, 11), (5, 7), (11, 17)]

# Function to calculate slope between two points
def calculate_slope(p1, p2):
    (x1, y1) = p1
    (x2, y2) = p2
    if x1 == x2:
        return float('inf')  # Vertical lines have infinite slope
    return (y2 - y1) / (x2 - x1)

# Calculate slopes for all pairs of points
slopes = []

for i in range(len(points)):
    for j in range(i + 1, len(points)):
        slope = calculate_slope(points[i], points[j])
        slopes.append((points[i], points[j], slope))

# Find which three points have the same slope for different pairs
three_points_on_same_line = []

for i in range(len(points)):
    for j in range(i + 1, len(points)):
        for k in range(j + 1, len(points)):
            slope_ij = calculate_slope(points[i], points[j])
            slope_jk = calculate_slope(points[j], points[k])
            slope_ik = calculate_slope(points[i], points[k])
            
            if sp.simplify(slope_ij - slope_jk) == 0 and sp.simplify(slope_jk - slope_ik) == 0:
                three_points_on_same_line = [points[i], points[j], points[k]]

# Find the point that is not on the same line
point_not_on_same_line = [point for point in points if point not in three_points_on_same_line][0]
print(point_not_on_same_line)
