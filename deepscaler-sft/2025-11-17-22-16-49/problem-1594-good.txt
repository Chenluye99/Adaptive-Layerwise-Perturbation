def to_base_4(n):
    if n == 0:
        return "0"
    digits = []
    while n:
        digits.append(int(n % 4))
        n //= 4
    return ''.join(str(x) for x in digits[::-1])

# Step 1: Convert the numbers from base 4 to base 10
