import sympy

# Step 1: Define the function to compute tau(n)
def tau(n):
    return sympy.divisor_count(n)

# Step 2: Define the function to compute S(n)
def compute_S(n):
    S_n = 0
    for i in range(1, n + 1):
        S_n += tau(i)
    return S_n
