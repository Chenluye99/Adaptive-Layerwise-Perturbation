import math

def find_smallest_n_for_perfect_square(a_start):
    a_n = a_start
    n = 1

    while True:
        sqrt_a_n = math.isqrt(a_n)  # Get the greatest integer less than or equal to sqrt(a_n)
        a_n_plus_1 = a_n + 2 * sqrt_a_n
        if math.isqrt(a_n_plus_1) ** 2 == a_n_plus_1:
            return n
        a_n = a_n_plus_1
        n += 1

# Part 1: a = 8
result_a_8 = find_smallest_n_for_perfect_square(8)

# Part 2: a = 2017
result_a_2017 = find_smallest_n_for_perfect_square(2017)
