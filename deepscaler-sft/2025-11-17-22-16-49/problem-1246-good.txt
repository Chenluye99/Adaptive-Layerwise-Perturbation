import sympy as sp

# Define the vertices of the square
A = (0, 0)
B = (7, 0)
C = (7, 7)
D = (0, 7)

# Define the vertices of the smaller regions
W = (3, 0)
X = (7, 3)
Y = (4, 7)
Z = (0, 4)

# Function to calculate the signed area of a polygon given its vertices
def polygon_area(vertices):
    n = len(vertices)
    sum1 = sum(vertices[i][0] * vertices[(i + 1) % n][1] for i in range(n))
    sum2 = sum(vertices[i][1] * vertices[(i + 1) % n][0] for i in range(n))
    return 0.5 * abs(sum1 - sum2)

# Vertices of the shaded regions
# Shaded regions formed by vertices of the square and the smaller regions
shaded_region_1 = [A, W, Z]
shaded_region_2 = [W, B, X]
shaded_region_3 = [X, C, Y]
shaded_region_4 = [Y, D, Z]

# Get the areas of each of these shaded regions
area_1 = polygon_area(shaded_region_1)
area_2 = polygon_area(shaded_region_2)
area_3 = polygon_area(shaded_region_3)
area_4 = polygon_area(shaded_region_4)

# Total shaded area is the sum of the areas of these four regions
total_shaded_area = area_1 + area_2 + area_3 + area_4
print(total_shaded_area)
