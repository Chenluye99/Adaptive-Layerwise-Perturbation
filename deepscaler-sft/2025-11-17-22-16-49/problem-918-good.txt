\[ \frac{S}{2} = \sum_{k=1}^{\infty} \frac{k}{2^{k+1}} = \frac{1}{2^2} + \frac{2}{2^3} + \frac{3}{2^4} + \frac{4}{2^5} + \cdots \]

\[ S - \frac{S}{2} = \left( \frac{1}{2^1} + \frac{2}{2^2} + \frac{3}{2^3} + \frac{4}{2^4} + \cdots \right) - \left( \frac{1}{2^2} + \frac{2}{2^3} + \frac{3}{2^4} + \frac{4}{2^5} + \cdots \right) \]

\[ S - \frac{S}{2} = \frac{1}{2} + \frac{1}{2^2} + \frac{1}{2^3} + \frac{1}{2^4} + \cdots \]
