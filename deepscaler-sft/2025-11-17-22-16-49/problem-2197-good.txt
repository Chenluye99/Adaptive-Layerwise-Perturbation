\[ n^2 = x^2 + y^2 + z^2 + 2xy + 2yz + 2zx + 3x + 3y + 3z - 6. \]

\[ n^2 = (x + y + z)^2 + 3(x + y + z) - 6. \]

\[ n^2 = s^2 + 3s - 6. \]

\[ n^2 = s^2 + 3s - 6. \]

for s in range(1, 1000):  # We choose an upper bound of 1000 for checking purposes.

    if value >= 0:

            if found:
