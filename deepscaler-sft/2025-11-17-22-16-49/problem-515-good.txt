```python
import sympy as sp

# Define variables
a = sp.symbols('a', real=True)

# Define the function f(x)
x = sp.symbols('x', real=True)
f = 2 * sp.sin(x) * (sp.sin(x) + sp.cos(x))

# Substitute x = a/2 into f(x)
f_sub = f.subs(x, a / 2)

# Set the equation f(a/2) = 1 + 3*sqrt(2)/5
equation = sp.Eq(f_sub, 1 + 3 * sp.sqrt(2) / 5)

# Solve for a in the interval (3pi/4, 5pi/4)
solution = sp.solveset(equation, a, domain=sp.Interval(3 * sp.pi / 4, 5 * sp.pi / 4))

# Get the valid solution for a
a_value = next(iter(solution))

# Find cos(a)
cos_a = sp.cos(a_value)

print(cos_a)
```
