import itertools

def find_nm_pairs():
    for n in range(3, 100):  # Search for reasonable upper limits for n and m (start with 3 as given in problem statement and continue up to an upper limit of 100 by reasonable guess)
        for m in range(3, 100):  # Do the same for m
            count_handshakes = 0
            for i in range(n):
                for j in range(m):
                    neighbors = []
                    if i > 0:
                        neighbors.append((i - 1, j))
                    if i < n - 1:
                        neighbors.append((i + 1, j))
                    if j > 0:
                        neighbors.append((i, j - 1))
                    if j < m - 1:
                        neighbors.append((i, j + 1))
                    
                    for n1, n2 in neighbors:
                        if 0 <= n1 < n and 0 <= n2 < m:
                            count_handshakes += 1
                        
            if count_handshakes == 252:
                return n, m, n * m

result = find_nm_pairs()
