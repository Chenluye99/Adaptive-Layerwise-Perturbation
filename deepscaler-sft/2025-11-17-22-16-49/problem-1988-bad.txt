```python
import sympy as sp

x, y, a, b = sp.symbols('x y a b')

# Define the equations
eq1 = 6 * x - 4 * y - a
eq2 = 6 * y - 9 * x - b

# Solve the system of equations for a and b in terms of x and y
solution = sp.solve([eq1, eq2], (a, b))

# Get the expressions for a and b
a_expr = solution[a]
b_expr = solution[b]

# Find the ratio a/b
ratio = a_expr / b_expr
print(ratio)
```
