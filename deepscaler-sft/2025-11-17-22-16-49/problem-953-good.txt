# Function to calculate area using shoelace formula (a method to determine the area of a polygon when the coordinates of the vertices are known)
def polygon_area(vertices):
    n = len(vertices)
    sum1 = sum(vertices[i][0] * vertices[(i + 1) % n][1] for i in range(n))
