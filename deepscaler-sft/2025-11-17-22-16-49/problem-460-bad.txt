def is_condition_met(num_people, super_knowers):
    # Create an adjacency list to represent who knows who
    graph = {i: set() for i in range(num_people)}
    for sk in super_knowers:
        for j in range(num_people):
            if j != sk:
                graph[sk].add(j)
    
    # Check if for any group of 4 people, there is someone who knows all three others
    for comb in itertools.combinations(range(num_people), 4):
        knows_all_others = False
        for person in comb:
            if all(graph[person].issuperset(set(comb) - {person})):
                knows_all_others = True
                break
        if not knows_all_others:
            return False
    return True

min_super_knowers = None
