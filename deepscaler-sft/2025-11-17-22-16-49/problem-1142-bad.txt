   \[
   \frac{m}{6} + 5 \geq 20

   \[
   \frac{m}{6} + 5 \geq 20

   \[
   \frac{m}{6} \geq 15
   \]

# Let's verify this step-by-step by coding the conditions in Python.

# We start checking from m = 90 and go upwards until we ensure that each tray has at least 20 baked goods after adding 5 croissants to each tray that originally got m/6 muffins.

def find_min_muffins():

    while True:
        muffins_per_tray = m / 6
        total_goods_per_tray = muffins_per_tray + 5

        if total_goods_per_tray >= 20:

min_muffins = find_min_muffins()
