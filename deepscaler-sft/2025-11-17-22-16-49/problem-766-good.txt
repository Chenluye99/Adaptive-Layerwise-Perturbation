```python
# Define the points P, Q, R

Q = (8, 8, 1)
R = (11, 3, 9)

# Function to compute Euclidean distance between two points
def distance(point1, point2):
    return sp.sqrt(sum((x - y) ** 2 for x, y in zip(point1, point2)))

# Calculate distances between P, Q, and R
d_PQ = distance(P, Q)
d_QR = distance(Q, R)
d_PR = distance(P, R)

print(f"Distance between P and Q: {d_PQ}")
print(f"Distance between Q and R: {d_QR}")
print(f"Distance between P and R: {d_PR}")
```
