```python
def is_palindrome(n):
    """Check if a number is a palindrome."""
    return str(n) == str(n)[::-1]

# Start checking from 1 and go upwards
num = 40305
i = 1

while True:
    if is_palindrome(num + i):
        result = i
        break
    i += 1

print(result)
```
