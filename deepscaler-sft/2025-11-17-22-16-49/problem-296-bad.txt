# We will place the 5 boys at positions 0, 5, 10, 15, and 20

        count_5m_away_boys = 0
        for boy_pos in boys_positions:
            if abs(i - boy_pos) == 5:
                count_5m_away_boys += 1
        
        if count_5m_away_boys == 2:
            if i not in found_positions:

    if len(found_positions) == num_girls:  # If we found 'num_girls' positions all of which satisfy the condition, we keep track of this
