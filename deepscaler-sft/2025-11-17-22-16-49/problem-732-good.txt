# Function to convert a base-10 number to base-6 and get its units digit
def to_base_6(num):
    if num == 0:
        return "0"
    digits = []
    while num:
        digits.append(int(num % 6))
        num //= 6
    return ''.join(str(x) for x in digits[::-1])

# Numbers in base 6
num1_base_6 = "413"
num2_base_6 = "215"

# Convert the base-6 numbers to base-10
num1_base_10 = int(num1_base_6, 6)

# Perform the subtraction in base-10
result_base_10 = num1_base_10 - num2_base_10

# Convert the result back to base-6
result_base_6 = to_base_6(result_base_10)

# Get the units digit of the result in base-6
units_digit = result_base_6[-1]
print(units_digit)
