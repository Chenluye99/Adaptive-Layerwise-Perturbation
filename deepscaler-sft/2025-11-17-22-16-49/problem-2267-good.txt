import itertools
import numpy as np

# Get all permutations of a sequence of 6 distinct objects
num_beads = 6
beads = list(range(num_beads))
permutations = list(itertools.permutations(beads, num_beads))

# Define a function to get all possible placements in the 2x3 grid
def get_grid_permutations():
    return permutations

# Define a function to generate all unique arrangements considering rotations and reflections as the same
def get_unique_arrangements(grid_permutations):
    unique_arrangements = set()
    
    for p in grid_permutations:
        grid = np.array(p).reshape((2, 3))
        
        # Generate all rotations and reflections of the grid and get the canonical form (the lexicographically smallest form in the set of all rotations and reflections)
        all_transformed_grids = set()
        
        for i in range(4):  # Rotate 0, 90, 180, 270 degrees
            rotated_grid = np.rot90(grid, i)
            all_transformed_grids.add(tuple(map(tuple, rotated_grid)))
        
        reflected_grid = np.fliplr(grid)
        for i in range(4):  # Rotate 0, 90, 180, 270 degrees
            rotated_reflected_grid = np.rot90(reflected_grid, i)
            all_transformed_grids.add(tuple(map(tuple, rotated_reflected_grid)))
        
        # Get the lexicographically smallest form of all transformed grids
        canonical_form = min(all_transformed_grids, key=lambda x: str(x))
        unique_arrangements.add(canonical_form)
    
    return len(unique_arrangements)

grid_permutations = get_grid_permutations()
num_unique_arrangements = get_unique_arrangements(grid_permutations)

print(num_unique_arrangements)
