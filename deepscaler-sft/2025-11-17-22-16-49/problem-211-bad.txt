# We will model this as a graph problem and use networkx library to check for cycles
import networkx as nx

def find_minimum_flights_for_cycle(n):
    min_flights = 0
    
    for num_flights in range(1, 100):  # Arbitrarily decide to check up to 100 flights
        found = True
        
        # Create all possible ways to connect 13 nodes with num_flights
        nodes = list(range(n))
        
        for edge_list in itertools.combinations([(i, j) for i in nodes for j in nodes if i != j], num_flights):

            G.add_nodes_from(nodes)
            G.add_edges_from(edge_list)
            
            # Check if the graph formed has a cycle
            if not any(nx.has_path(G, source, target) and source != target for source in nodes for target in nodes):
                found = False
                break
                
        if found:

            break
    
    return min_flights
