import itertools

# List of 5 keys, where 'H' is the house key and 'C' is the car key
keys = ['H', 'C', 'K1', 'K2', 'K3']

# Function to check if two arrangements are equivalent under rotation and reflection for a keychain
def are_equivalent(arr1, arr2):
    n = len(arr1)
    for i in range(n):
        rotated = arr1[i:] + arr1[:i]
        if rotated == arr2 or rotated[::-1] == arr2:  # Check both rotation and reflection
            return True
    return False

# Generate all distinct ways to arrange the keys on a keychain and then check the condition that 'H' and 'C' must be next to each other
distinct_arrangements = set()

for perm in set(itertools.permutations(keys)):
    # Create a list to represent the keychain by appending the first element to the end, to account for circular permutation
    chain = list(perm) + [perm[0]]
    
    # Normalize by rotating the list so that 'H' is as close to the start as possible (this is to avoid counting rotations of the same chain multiple times)
    min_rot = min([chain[i:] + chain[:i] for i in range(len(chain))], key=lambda x: (x.index('H'), x.index('C')))
    if abs(min_rot.index('H') - min_rot.index('C')) == 1 or abs(min_rot.index('H') - min_rot.index('C')) == len(min_rot) - 1:
        distinct_arrangements.add(tuple(min_rot))

# Now filter out only those arrangements where 'H' and 'C' are next to each other
count = 0
for arrangement in distinct_arrangements:
    chain = list(arrangement) + [arrangement[0]]
    if abs(chain.index('H') - chain.index('C')) == 1 or abs(chain.index('H') - chain.index('C')) == len(chain) - 1:
