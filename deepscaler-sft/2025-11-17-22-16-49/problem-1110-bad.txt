# Step 1: Generate all possible arrangements of the lamps and all ways to turn 3 of them on

# Create a list of 3 red and 3 blue lamps
lamps = ['R', 'R', 'R', 'B', 'B', 'B']

# Get all unique ways to arrange 3 red and 3 blue lamps in a row

count_successful_outcomes = 0
total_outcomes = 0

    for lamps_to_turn_on in itertools.combinations(range(6), 3):

        arrangement_copy = list(arrangement)
        
        # Turn on the selected 3 lamps
        for index in lamps_to_turn_on:
            arrangement_copy[index] = 'O' if arrangement_copy[index] == 'R' or arrangement_copy[index] == 'B' else 'O'
        
        # Check the conditions
        leftmost_lamp = arrangement_copy[0]
        rightmost_lamp = arrangement_copy[5]
