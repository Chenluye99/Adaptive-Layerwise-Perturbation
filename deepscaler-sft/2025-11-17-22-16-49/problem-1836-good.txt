import math
from functools import reduce

# Function to calculate LCM of two numbers
def lcm(x, y):
    return x * y // math.gcd(x, y)

# Function to calculate LCM of a list of numbers
def lcm_multiple(numbers):
    return reduce(lcm, numbers)

# Calculate LCM of numbers from 1 to 25
lcm_1_to_25 = lcm_multiple(range(1, 26))

# Function to check if a number is only divisible by all numbers from 1 to 25, except for two consecutive numbers
def check_number(n):
    divisible = True
    for i in range(1, 26):
        if n % i != 0:
            divisible = False
            break
    return divisible

min_number = float("inf")

# Check all possible pairs of two consecutive numbers to omit
for i in range(1, 25):
    # Create a list of numbers from 1 to 25 excluding the two consecutive numbers i and i+1
    numbers_to_check = [num for num in range(1, 26) if num not in [i, i + 1]]
    
    # Calculate the LCM of the numbers excluding i and i+1
    lcm_excluding_i_i1 = lcm_multiple(numbers_to_check)
    
    # Check if this is the smallest such number
    if all(lcm_excluding_i_i1 % num == 0 for num in numbers_to_check):
        if lcm_excluding_i_i1 < min_number:

print(min_number)
