```python
# Initialize the first few rows of the triangle as per the problem statement
triangle = []

# Generate the triangle up to the 100th row
for n in range(100):
    if n == 0:
        row = [0]
    elif n == 1:
        row = [1, 1]
    else:
        row = [i for i in range(n + 1)]
        for i in range(2, n):
            row[i] = triangle[n - 1][i - 1] + triangle[n - 1][i]
    triangle.append(row)

# Get the 100th row (index 99 because Python uses 0-based indexing)
hundredth_row = triangle[99]

# Calculate the sum of the 100th row
sum_of_hundredth_row = sum(hundredth_row)
print(sum_of_hundredth_row)
```
