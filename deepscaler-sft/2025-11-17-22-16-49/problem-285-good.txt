import sympy as sp

# Define the variables
x, y = sp.symbols('x y')

# Define the constraint equation
constraint = x**2 - sp.sqrt(2) * x + y**2 - 4

# Define the function to maximize: AB + AC
f = sp.sqrt(x**2 + y**2) + sp.sqrt((x - sp.sqrt(2))**2 + y**2)

# Use Lagrange multipliers to solve for the maximum value
lambda_ = sp.symbols('lambda')
L = f - lambda_ * constraint

# Take partial derivatives of L with respect to x, y, and lambda, set them to 0
partial_x = sp.diff(L, x)
partial_y = sp.diff(L, y)
partial_lambda = sp.diff(L, lambda_)

# Solve the system of equations
solutions = sp.solve([partial_x, partial_y, partial_lambda], (x, y, lambda_))

# Evaluate f at the solutions to find the maximum value
max_value = 0
for sol in solutions:
    x_val, y_val, _ = sol
    if x_val.is_real and y_val.is_real:
        value = f.subs({x: x_val, y: y_val})
        if value > max_value:

print(max_value)
