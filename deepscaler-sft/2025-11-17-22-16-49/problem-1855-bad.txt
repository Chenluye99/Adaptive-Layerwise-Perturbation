```python
import itertools

# Let's label the animals for clarity
males = ['male_lion', 'male_tiger', 'male_elephant', 'male_zebra']
females = ['female_lion', 'female_tiger', 'female_elephant', 'female_zebra']

# Combine the animals and ensure that the male giraffe is always the first one fed
animals = males + females
num_ways = 0

# Fix the male giraffe as the first animal
fixed_first_animal = 'male_giraffe'
animals_and_fixed_first = [fixed_first_animal] + [animal for animal in animals if animal != 'male_giraffe']

# Generate all permutations of the remaining 7 animals
for perm in list(itertools.permutations(animals_and_fixed_first)):
    current_gender = 'male'  # It starts with 'male_giraffe', so the current gender is male
    
    # We need to track the sequence based on our rule that each time an animal is fed, the next one must be of the opposite gender
    sequence_check = [fixed_first_animal]
    
    for animal in perm:
        if current_gender == 'male':
            if animal in females:
                sequence_check.append(animal)
                current_gender = 'female'
            else:
                # This violates the gender alternation rule, so we break and continue checking other permutations
                break
        elif current_gender == 'female':
            if animal in males:
                sequence_check.append(animal)
                current_gender = 'male'
            else:
                # This violates the gender alternation rule, so we break and continue checking other permutations
                break
    
    # If by the end of a sequence check, we have exactly 8 animals (4 males and 4 females plus the fixed male giraffe), then it's a valid way
    if len(sequence_check) == 8:
        num_ways += 1

print(num_ways)
```
