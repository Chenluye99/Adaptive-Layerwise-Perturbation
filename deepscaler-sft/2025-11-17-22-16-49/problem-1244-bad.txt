```python
import math

def g(x):
    count = 0
    while x % 2 == 0:
        x //= 2
        count += 1
    return 2 ** count

# Now we will calculate S_n for different values of n and check if S_n is a perfect square

max_n = 999
perfect_square_n = []

for n in range(1, max_n + 1):
    S_n = sum(g(2 * k) for k in range(1, 2 ** (n - 1) + 1))
    sqrt_S_n = int(math.isqrt(S_n))
    if sqrt_S_n * sqrt_S_n == S_n:
