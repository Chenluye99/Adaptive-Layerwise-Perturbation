# Define the coordinates of the points
A = (0, 3)
B = (0, 0)
C = (4, 0)

# Find the coordinates of G, the centroid of triangle ABC
G = ((A[0] + B[0] + C[0]) / 3, (A[1] + B[1] + C[1]) / 3)

# Function to find the projection of G onto any line passing through two points
def project_onto_line(point, line_start, line_end):
    x0, y0 = point
    x1, y1 = line_start
    x2, y2 = line_end
    
    # Get the slope and intercept of the line passing through line_start and line_end
    if x2 != x1:
        m = (y2 - y1) / (x2 - x1)
        b = y1 - m * x1
    else:
        m = float("inf")
        b = x1
    
    # Find the perpendicular line passing through G and get its intersection with the line of the edge
    if m == float("inf"):
        x_intersect = x0
        y_intersect = b
    else:
        m_perpendicular = -1 / m
        b_perpendicular = y0 - m_perpendicular * x0
        
        # Get intersection of perpendicular line with the edge
        x_intersect = (b - b_perpendicular) / (m_perpendicular - m)
        y_intersect = m * x_intersect + b
    
    return (x_intersect, y_intersect)
