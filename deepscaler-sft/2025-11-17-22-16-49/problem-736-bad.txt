def sum_of_divisors(n):
    """Compute the sum of the positive integer divisors of n."""
    return sum(i for i in range(1, n + 1) if n % i == 0)

def is_bouncy_prime(n):
    """Check if n is a bouncy prime."""
    f_n = sum_of_divisors(n)
    f_f_n = sum_of_divisors(f_n)
