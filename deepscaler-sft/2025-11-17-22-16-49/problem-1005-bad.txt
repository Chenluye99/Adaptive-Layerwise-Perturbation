1. Define the function \( f(x, y, z) = \frac{4}{x} + \frac{9}{y} + \frac{16}{z} \).
2. Define the constraint \( g(x, y, z) = x + y + z - 3 = 0 \).

import numpy as np
from scipy.optimize import minimize

# Define the function to minimize
def objective(vars):
    x, y, z = vars
    return 4/x + 9/y + 16/z

# Define the constraint function
def constraint(vars):
    x, y, z = vars
    return x + y + z - 3

# Define bounds for x, y, z (they must be positive)
bounds = [(1e-6, None), (1e-6, None), (1e-6, None)]

# Define initial guess for x, y, z
initial_guess = [1, 1, 1]

# Define constraints in the form required by `minimize`
constraint_eq = {'type': 'eq', 'fun': constraint}

# Perform the optimization
solution = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=[constraint_eq])

# Extract the optimized values of x, y, z and the minimum value of the function
x_opt, y_opt, z_opt = solution.x
min_value = solution.fun

print((x_opt, y_opt, z_opt, min_value))
