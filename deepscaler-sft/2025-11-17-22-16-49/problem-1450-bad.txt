        return sp.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)
    
    # Get all six distances between the four points
    distances = [distance(points[i], points[j]) for i in range(4) for j in range(i+1, 4)]

    if distances[0] == distances[1] == distances[2] == distances[3] and distances[4] == distances[5]:
        side_length = distances[0]

        return sp.simplify(diagonal_length - sp.sqrt(2) * side_length) == 0
