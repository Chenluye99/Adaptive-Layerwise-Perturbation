import sympy as sp

# Define the coordinates
A = (0, 3)
B = (0, 0)
C = (4, 0)

# Calculate the centroid G
G = ((A[0] + B[0] + C[0]) / 3, (A[1] + B[1] + C[1]) / 3)

# Function to find the projection of a point onto a line
def projection_point_to_line(point, line_point1, line_point2):
    x1, y1 = point
    x2, y2 = line_point1
    x3, y3 = line_point2
    
    # Vector along the line
    v = (x3 - x2, y3 - y2)
    # Vector from line point 1 to the point
    w = (x1 - x2, y1 - y2)
    
    # Project w onto v
    projection_length = (w[0] * v[0] + w[1] * v[1]) / (v[0] ** 2 + v[1] ** 2)
    proj = (x2 + projection_length * v[0], y2 + projection_length * v[1])
    
    # Find the perpendicular distance from G to the line BC, AC, AB
    return proj

# Get the projections
P = projection_point_to_line(G, B, C)
Q = projection_point_to_line(G, C, A)
R = projection_point_to_line(G, A, B)

# Calculate distances GP, GQ, GR
GP = sp.sqrt((G[0] - P[0])**2 + (G[1] - P[1])**2)
GQ = sp.sqrt((G[0] - Q[0])**2 + (G[1] - Q[1])**2)
GR = sp.sqrt((G[0] - R[0])**2 + (G[1] - R[1])**2)

# Get the sum of these distances
