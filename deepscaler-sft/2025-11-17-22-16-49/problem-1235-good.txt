import numpy as np

# Vertices of a regular hexagon centered at origin with side length 1
hexagon_vertices = [(np.cos(2 * np.pi * k / 6), np.sin(2 * np.pi * k / 6)) for k in range(6)]

# Get the coordinates for A, B, E, and F
A = hexagon_vertices[0]
B = hexagon_vertices[1]
E = hexagon_vertices[4]
F = hexagon_vertices[5]

# Extend AB and EF to meet at P
# Parametric equation for the line AB: (x, y) = A + t * (B - A) for t >= 0
# Parametric equation for the line EF: (x, y) = E + s * (F - E) for s >= 0

# Find the intersection point P of the lines AB and EF by setting the equations equal:
# A + t * (B - A) = E + s * (F - E)

# Get direction vectors
dir_AB = np.array(B) - np.array(A)
dir_EF = np.array(F) - np.array(E)

# Set up the equations to solve for t and s
A = np.array(A)
E = np.array(E)

# We need to solve (A + t * dir_AB) = (E + s * dir_EF)
# This translates to a system of linear equations:
# t * dir_AB[0] + A[0] = s * dir_EF[0] + E[0]
# t * dir_AB[1] + A[1] = s * dir_EF[1] + E[1]

coeff_matrix = np.array([dir_AB, -dir_EF]).T
rhs = E - A

# Solving the system of equations
solution = np.linalg.solve(coeff_matrix, rhs)

t, s = solution

# Get the intersection point P
P = A + t * dir_AB

# Now, get the angles of vectors AP and EP
AP = P - A
EP = P - E

# Calculate the angle between vectors AP and EP using dot product formula
dot_product = np.dot(AP, EP)
magnitude_AP = np.linalg.norm(AP)
magnitude_EP = np.linalg.norm(EP)

cos_theta = dot_product / (magnitude_AP * magnitude_EP)
angle_rad = np.arccos(cos_theta)

# Convert the angle from radians to degrees
angle_deg = np.degrees(angle_rad)
print(angle_deg)
