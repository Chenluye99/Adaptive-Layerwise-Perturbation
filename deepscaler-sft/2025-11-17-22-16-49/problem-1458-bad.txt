# Function to check if angle APB is obtuse
def is_angle_obeus(P, A, B):
    # Vector AB
    AB = (B[0] - A[0], B[1] - A[1])
    # Vector AP
    AP = (P[0] - A[0], P[1] - A[1])
    # Vector BP
    BP = (P[0] - B[0], P[1] - B[1])
    
    # Dot products
    dot_product_AB_AP = AB[0] * AP[0] + AB[1] * AP[1]
    dot_product_AB_BP = AB[0] * BP[0] + AB[1] * BP[1]
