def triangle_area(A, P, Q):
    x1, y1 = A
    x2, y2 = P
    x3, y3 = Q
    return sp.Rational(1, 2) * abs(x1 * (y2 - y3) + x2 * (y3 - y1) + x3 * (y1 - y2))

print(area_APQ)
