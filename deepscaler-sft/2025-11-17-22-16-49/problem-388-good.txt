# We will iterate through all possible configurations of the rectangle that would give 40 squares and ensure it has more than one row. Then for each configuration, we will calculate how many squares were not colored by Andrew.

def find_uncolored_squares():
    # We start checking from (2, 20) to (5, 8) and ensure the configurations meet the requirements of having more than one row.
    for rows in range(2, 41):
        cols = 40 // rows
        if rows * cols == 40:
            unc_colored_count = 0
            
            # For each valid configuration, we need to identify the middle row(s) and count the squares not colored by Andrew.
            if rows == 2:
                middle_row = [rows - 1]
            elif rows % 2 == 0:
                middle_row = [rows // 2 - 1, rows // 2]
            else:
                middle_row = [rows // 2]
            
            squares_in_middle_row = len(middle_row) * cols
            
            unc_colored_count = 40 - squares_in_middle_row
            
            yield unc_colored_count

# Get the result by iterating through all valid configurations.
results = list(find_uncolored_squares())
print(results)
