# Define the expression to maximize: λ*xy + yz
expression = λ * x * y + y * z

# Use Lagrange multipliers method to find critical points
λ_mult = sp.symbols('λ_mult', real=True, positive=True)
lagrange_function = expression - λ_mult * constraint

# Take partial derivatives
partial_x = sp.diff(lagrange_function, x)
partial_y = sp.diff(lagrange_function, y)
partial_z = sp.diff(lagrange_function, z)
partial_λ_mult = sp.diff(lagrange_function, λ_mult)

solutions = sp.solve([partial_x, partial_y, partial_z, partial_λ_mult], (x, y, z, λ, λ_mult), dict=True)

# Find the maximum value of λ such that λ * xy + yz <= sqrt(5)/2 for all solutions
max_lambda = 0

    lambda_val = sol[λ]
    if λ * sol[x] * sol[y] + sol[y] * sol[z] <= sp.sqrt(5) / 2:
