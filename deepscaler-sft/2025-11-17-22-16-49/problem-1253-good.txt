import sympy as sp

# Radius of the circle
r = 12

# Central angle of 60 degrees in radians
theta = sp.rad(60)

# Area of one 60 degree sector
sector_area = (theta / (2 * sp.pi)) * sp.pi * r**2

# Now, the overlapping area would be the intersection of two such 60 degree sectors.
# We notice that each sector covers a 60-degree angular slice and when overlapping each such appropriately forms a intersection shape as noted as two halves forming the lens shape representing appropriately (shared overlapping part)

# By geometrical properties and knowing how overlapping areas (having symmetrical and identical shape respectively):
intersecting_overlap = 2 * (sector_area - (r**2 * sp.sin(theta) / 2))

print(intersecting_overlap)
